Ct.forBlock.detect_shape=function(i,r){r.definitions_.import_cv2="import cv2",r.definitions_.import_numpy="import numpy as np",r.definitions_.import_math="import math",r.definitions_.detect_shape_func=`
def detect_shape(contours_obj, index):
    # Извлекаем контуры из результата cv2.findContours
    if isinstance(contours_obj, tuple) and len(contours_obj) >= 1:
        contours = contours_obj[0]
    elif isinstance(contours_obj, (list, tuple)):
        contours = contours_obj
    else:
        return "неизвестно"
    
    if contours is None or len(contours) == 0:
        return "неизвестно"
    
    # Проверяем индекс
    index = int(index)
    if index < 0 or index >= len(contours):
        return "неизвестно"
    
    contour = contours[index]
    
    # Аппроксимируем контур
    epsilon = 0.02 * cv2.arcLength(contour, True)
    approx = cv2.approxPolyDP(contour, epsilon, True)
    
    # Количество вершин
    vertices = len(approx)
    
    # Площадь контура
    area = cv2.contourArea(contour)
    
    # Периметр контура
    perimeter = cv2.arcLength(contour, True)
    
    # Отношение площади к квадрату периметра (для круга это ~1/(4*pi) ≈ 0.08)
    if perimeter > 0:
        circularity = 4 * math.pi * area / (perimeter * perimeter)
    else:
        circularity = 0
    
    # Определяем форму
    if vertices == 3:
        return "треугольник"
    elif vertices == 4:
        # Квадрат или прямоугольник
        x, y, w, h = cv2.boundingRect(approx)
        aspect_ratio = float(w) / h if h > 0 else 0
        
        # Если соотношение сторон близко к 1, это квадрат
        if 0.95 <= aspect_ratio <= 1.05:
            return "квадрат"
        else:
            return "прямоугольник"
    elif vertices == 5:
        return "пятиугольник"
    elif vertices == 6:
        return "шестиугольник"
    elif vertices > 6:
        # Много вершин - может быть круг или многоугольник
        if circularity > 0.7:
            return "круг"
        else:
            return "многоугольник"
    else:
        return "неизвестно"

def find_poses(img, det_boxes):
    """Run model inference on *img* and return a structured result object.

    Returns a dict with:
        - poses: list of pose dicts, each with keys
            - bbox: [x1, y1, x2, y2] in *img* pixel coordinates
            - kpts: numpy.ndarray of shape (17, 3) with x, y, conf
        - total_dets: total number of raw detections returned by the model
    """
    ih, iw = img.shape[:2]

    # Factors to convert network coordinates → input frame coordinates
    sx = iw / float(IMG_SIZE[0])
    sy = ih / float(IMG_SIZE[1])

    poses = []
    for box in det_boxes:
        x1, y1, x2, y2 = float(box.xmin), float(box.ymin), float(box.xmax), float(box.ymax)

        # Convert key-points to an array and skip invalid ones
        karr = get_kpts_array(np.asarray(box.keypoint))
        if karr is None:
            continue

        # Scale key-points and bbox to the original frame size
        karr = karr.copy()
        karr[:, 0] *= sx
        karr[:, 1] *= sy
        bbox_scaled = [x1 * sx, y1 * sy, x2 * sx, y2 * sy]

        poses.append({
            'bbox': bbox_scaled,
            'kpts': karr,
        })

    return {
        'poses': poses,
        'total_dets': len(det_boxes),
    }
